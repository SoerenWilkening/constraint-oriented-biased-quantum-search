**tests/test_constraint.c** -- Add a new test function `test_incremental_vs_full_recalc` that:

1. Builds a multi-constraint problem (e.g., 3 constraints, 8 variables with mixed positive/negative coefficients)
2. Creates a solution state and sets some bits
3. Calls preprocessing() on the constraints (required for adjusted_constraint_violation)
4. Computes violation using full-recalc: `constraint_violation(con, sol, cnstr)` for each constraint
5. Computes violation using incremental path:
   - `prepare_constraints(con, sol, &ful_con)` to get baseline
   - For each variable flip: use `adjusted_constraint_violation()` with positive and negative indices
   - Compute `remainings[cnstr] - totals[cnstr]` to get the incremental result
6. Asserts exact int64_t equality between the two approaches for each constraint

The test should cover:
- Multiple constraint types (different signs, different RHS values)
- Both single-flip and multi-flip moves
- Cases where flipping changes feasibility status
- The specific pattern used in explore_neighbourhood (flip bits, compute, unflip)

Example structure:
```c
static void test_incremental_vs_full_recalc(void **state) {
    (void)state;
    /* Build constraints: 3 constraints, 8 variables */
    int n = 8;
    new_constraints_t con = init_new_constraint();

    /* Constraint 0: 3*x0 + 2*x1 - 1*x2 + 4*x3 <= 6 */
    int64_t coeffs0[] = {3, 2, -1, 4, 0, 0, 0, 0};
    expression_t *e0 = build_linear_expression(n, coeffs0, LOWER, 6);
    add_expression_to_constraints(&con, e0);
    free_expression(e0);

    /* Constraint 1: -2*x0 + 5*x1 + 1*x4 + 3*x5 <= 8 */
    int64_t coeffs1[] = {-2, 5, 0, 0, 1, 3, 0, 0};
    expression_t *e1 = build_linear_expression(n, coeffs1, LOWER, 8);
    add_expression_to_constraints(&con, e1);
    free_expression(e1);

    /* Constraint 2: 1*x2 + 1*x3 + 1*x6 + 1*x7 <= 3 */
    int64_t coeffs2[] = {0, 0, 1, 1, 0, 0, 1, 1};
    expression_t *e2 = build_linear_expression(n, coeffs2, LOWER, 3);
    add_expression_to_constraints(&con, e2);
    free_expression(e2);

    /* Preprocess for incremental access */
    preprocessing(n, &con);

    /* Create solution: x0=1, x1=1, x3=1, x5=1 */
    state_t *sol = create_state(n);
    sw_setbit(sol->vector, 0);
    sw_setbit(sol->vector, 1);
    sw_setbit(sol->vector, 3);
    sw_setbit(sol->vector, 5);

    uint32_t C = con.num_constraints;

    /* Full-recalc baseline */
    int64_t *full_recalc = malloc(C * sizeof(int64_t));
    for (uint32_t i = 0; i < C; ++i)
        full_recalc[i] = constraint_violation(&con, sol, i);

    /* Incremental baseline */
    int64_t *remainings = malloc(C * sizeof(int64_t));
    for (uint32_t i = 0; i < C; ++i)
        remainings[i] = constraint_violation(&con, sol, i);

    /* Baseline should match */
    for (uint32_t i = 0; i < C; ++i)
        assert_int_equal(full_recalc[i], remainings[i]);

    /* Test single flip of variable 2 */
    sw_flpbit(sol->vector, 2); /* flip x2 */

    /* Full-recalc after flip */
    int64_t *full_after = malloc(C * sizeof(int64_t));
    for (uint32_t i = 0; i < C; ++i)
        full_after[i] = constraint_violation(&con, sol, i);

    /* Incremental after flip */
    array_t ful_con = sw_init(con.total_clauses);
    /* Need to prepare with ORIGINAL solution, so unflip, prepare, reflip */
    sw_flpbit(sol->vector, 2); /* back to original */
    prepare_constraints(&con, sol, &ful_con);
    sw_flpbit(sol->vector, 2); /* flip again */

    int64_t *totals = calloc(C, sizeof(int64_t));
    array_t inv = sw_init(con.total_clauses);
    int *changed_con = calloc(MINSIZE, sizeof(int));
    int num_con_changes = 0;

    adjusted_constraint_violation(&con, 2,
        con.positive_indices, con.num_positive_indices,
        con.positive_offsets, sol, POSITIVE,
        totals, &ful_con, &changed_con, &num_con_changes, &inv);
    adjusted_constraint_violation(&con, 2,
        con.negative_indices, con.num_negative_indices,
        con.negative_offsets, sol, NEGATIVE,
        totals, &ful_con, &changed_con, &num_con_changes, &inv);

    /* Compare: incremental result = remainings[c] - totals[c] should equal full_after rhs - lhs */
    /* Actually, constraint_violation returns rhs - lhs. After flip, full_after[c] is the new rhs-lhs. */
    /* The incremental delta totals[c] adjusts the lhs, so the new constraint_violation = remainings[c] - totals[c]... */
    /* Wait -- need to verify the exact semantics. Let's just check they match. */
    for (uint32_t c = 0; c < C; ++c) {
        int64_t incremental_result = remainings[c] - totals[c];
        assert_int_equal(full_after[c], incremental_result);
    }

    /* Unflip and clean up */
    sw_flpbit(sol->vector, 2);
    free(changed_con);
    sw_clear(inv);
    sw_clear(ful_con);
    free(totals);
    free(full_after);
    free(full_recalc);
    free(remainings);
    free_state(sol, 1);
    free_constraints(&con);
}
```
