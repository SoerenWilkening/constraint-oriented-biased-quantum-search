Script structure:
```python
#!/usr/bin/env python3
"""Benchmark: incremental vs full-recalc constraint evaluation in local_search.

Run on any commit to measure local_search wall-clock time across problem sizes.
Compare results between commits to see before/after performance delta.

Usage:
    python benchmarks/benchmark_incremental.py
"""
import time
import statistics
import random
from cbqs import Model

CONFIGS = [
    # (num_vars, num_constraints, label)
    (10, 5, "small-10"),
    (20, 10, "small-20"),
    (50, 25, "medium-50"),
    (100, 50, "medium-100"),
    (200, 100, "large-200"),
    (500, 200, "large-500"),
]

NUM_RUNS = 5
STOPPING_TIME = 2.0  # seconds per run (cap to keep benchmark reasonable)
DISTANCE = 1         # single-flip neighbourhood


def build_problem(num_vars, num_constraints, seed=42):
    """Build a random knapsack-style problem."""
    rng = random.Random(seed)
    m = Model()

    # Add variables
    for i in range(num_vars):
        m.add_variable(f"x{i}")

    # Add random constraints: sum of random coefficients * x_i <= random rhs
    for c in range(num_constraints):
        expr = None
        for i in range(num_vars):
            coeff = rng.randint(-5, 10)
            if coeff != 0:
                term = m.get_variable(f"x{i}") * coeff
                expr = term if expr is None else expr + term
        if expr is not None:
            rhs = rng.randint(num_vars // 4, num_vars * 2)
            m.add_constraint(expr <= rhs)

    # Objective: maximize sum of random positive weights * x_i
    obj = None
    for i in range(num_vars):
        w = rng.randint(1, 20)
        term = m.get_variable(f"x{i}") * w
        obj = term if obj is None else obj + term
    m.set_objective(obj)

    m.set_param("stopping_time", STOPPING_TIME)
    m.set_param("distance", DISTANCE)

    return m


def benchmark_config(num_vars, num_constraints, label):
    """Run benchmark for one configuration."""
    times = []
    for run in range(NUM_RUNS):
        m = build_problem(num_vars, num_constraints, seed=42 + run)
        start = time.perf_counter()
        m.local_search()
        elapsed = time.perf_counter() - start
        times.append(elapsed)

    median_time = statistics.median(times)
    return {
        "label": label,
        "num_vars": num_vars,
        "num_constraints": num_constraints,
        "median_s": median_time,
        "min_s": min(times),
        "max_s": max(times),
        "all_times": times,
    }


def main():
    print("=" * 60)
    print("Incremental Evaluation Benchmark")
    print("=" * 60)
    print(f"Runs per config: {NUM_RUNS}")
    print(f"Stopping time: {STOPPING_TIME}s")
    print(f"Distance (neighbourhood): {DISTANCE}")
    print()

    results = []
    for num_vars, num_constraints, label in CONFIGS:
        print(f"Running {label} ({num_vars} vars, {num_constraints} constraints)...")
        r = benchmark_config(num_vars, num_constraints, label)
        results.append(r)
        print(f"  Median: {r['median_s']:.4f}s  Range: [{r['min_s']:.4f}, {r['max_s']:.4f}]")

    # Print markdown table
    print()
    print("## Results (Markdown)")
    print()
    print("| Config | Variables | Constraints | Median (s) | Min (s) | Max (s) |")
    print("|--------|-----------|-------------|------------|---------|---------|")
    for r in results:
        print(f"| {r['label']} | {r['num_vars']} | {r['num_constraints']} | "
              f"{r['median_s']:.4f} | {r['min_s']:.4f} | {r['max_s']:.4f} |")

    return results


if __name__ == "__main__":
    main()
```
