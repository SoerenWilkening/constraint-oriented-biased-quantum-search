  ```python
  def verify_solution(self):
      """Verify that the current solution satisfies all constraints
      and the reported objective value matches recomputation.

      Returns True if solution is valid, False otherwise.
      Emits UserWarning on violations but does not raise exceptions.
      Sets self._verified to the result.
      """
      import warnings

      # Check that solve() has been called
      if self.mod[0].global_opt is NULL:
          warnings.warn(
              "No solution to verify: solve() has not been called",
              UserWarning,
              stacklevel=2
          )
          self._verified = False
          return False

      verified = True

      # Create a state_py wrapper around the C global_opt state for eval
      # We need to access the global_opt state through the C model
      # The eval_con and eval_obj methods need a state_py object
      # Build one from the global_opt bits
      n = self.n
      bits = []
      for i in range(n):
          # Read bit i from global_opt state vector
          bits.append(sw_tstbit(self.mod[0].global_opt[0].vector, i))
      opt_state = state_py(self.mod[0].global_opt[0].tot_profit, bits)

      # Check constraint satisfaction using existing C evaluation
      constraints_ok = self.constraint.eval_con(opt_state)
      if not constraints_ok:
          warnings.warn(
              "Post-solve verification FAILED: solution violates one or more constraints",
              UserWarning,
              stacklevel=2
          )
          verified = False

      # Recompute objective value using existing C evaluation
      recomputed_obj = self.objective.eval_obj(opt_state)
      reported_obj = self.mod[0].global_opt[0].tot_profit * self.sense

      EPSILON = 1e-9
      if abs(recomputed_obj - reported_obj) > EPSILON:
          warnings.warn(
              f"Post-solve verification FAILED: reported objective {reported_obj} "
              f"does not match recomputed {recomputed_obj}",
              UserWarning,
              stacklevel=2
          )
          verified = False

      self._verified = verified
      return verified
  ```

  """Tests for RBST-02: Post-solve solution verification."""
  import warnings
  import pytest
  from cbqs.Model import Model
  from cbqs.Expression import Variable, Expression
  from cbqs.Constants import MAXIMIZE, MINIMIZE

  class TestVerifySolution:

      def _build_and_solve_knapsack(self, verify_flag=False):
          """Helper: build and solve a simple knapsack problem."""
          m = Model()
          xs = m.add_variables(5)
          x = [xs[i] for i in range(5)]
          m.add_constraint((x[0] + x[1] + x[2] + x[3] + x[4]) <= 3)
          m.set_objective(x[0] + x[1] + x[2] + x[3] + x[4], MAXIMIZE)
          m.close()
          m.solve(stopping_time=3, num_workers=1, verify=verify_flag)
          return m

      def test_verify_after_solve_returns_true(self):
          """A valid solved model should verify successfully."""
          m = self._build_and_solve_knapsack()
          result = m.verify_solution()
          assert result is True
          assert m._verified is True

      def test_verify_before_solve_warns(self):
          """Calling verify_solution() before solve() emits warning."""
          m = Model()
          xs = m.add_variables(3)
          m.add_constraint((xs[0] + xs[1] + xs[2]) <= 2)
          m.set_objective(xs[0] + xs[1] + xs[2], MAXIMIZE)
          m.close()
          with warnings.catch_warnings(record=True) as w:
              warnings.simplefilter("always")
              result = m.verify_solution()
              assert result is False
              assert m._verified is False
              assert len(w) >= 1
              assert "solve()" in str(w[0].message).lower() or "no solution" in str(w[0].message).lower()

      def test_verify_flag_on_solve(self):
          """solve(verify=True) sets _verified flag."""
          m = self._build_and_solve_knapsack(verify_flag=True)
          assert m._verified is not None
          # For a valid problem, should be True
          assert m._verified is True

      def test_verified_initially_none(self):
          """Before any verification, _verified is None."""
          m = Model()
          assert m._verified is None

      def test_verify_knapsack_constraints_satisfied(self):
          """Verify that solution satisfies the knapsack constraint."""
          m = self._build_and_solve_knapsack()
          result = m.verify_solution()
          assert result is True

      def test_verify_minimize_problem(self):
          """Verify works for MINIMIZE objective too."""
          m = Model()
          xs = m.add_variables(4)
          x = [xs[i] for i in range(4)]
          m.add_constraint((x[0] + x[1] + x[2] + x[3]) >= 2)
          m.set_objective(x[0] + x[1] + x[2] + x[3], MINIMIZE)
          m.close()
          m.solve(stopping_time=3, num_workers=1)
          result = m.verify_solution()
          # Result should be True for a valid solution
          assert isinstance(result, bool)

      def test_verify_multiple_constraints(self):
          """Verify with multiple constraints."""
          m = Model()
          xs = m.add_variables(4)
          x = [xs[i] for i in range(4)]
          m.add_constraint((x[0] + x[1]) <= 1)
          m.add_constraint((x[2] + x[3]) <= 1)
          m.set_objective(x[0] + x[1] + x[2] + x[3], MAXIMIZE)
          m.close()
          m.solve(stopping_time=3, num_workers=1)
          result = m.verify_solution()
          assert result is True
          assert m._verified is True
