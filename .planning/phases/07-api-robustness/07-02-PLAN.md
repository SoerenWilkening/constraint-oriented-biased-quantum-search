  """Tests for RBST-01: Input validation at Model level."""
  import pytest
  from cbqs.Model import Model
  from cbqs.Expression import Variable, Expression
  from cbqs.Constants import MAXIMIZE, MINIMIZE

  class TestAddConstraintValidation:

      def test_none_constraint_raises(self):
          m = Model()
          with pytest.raises(ValueError, match="cannot be None"):
              m.add_constraint(None)

      def test_expression_without_sense_raises(self):
          """Expression that has not been compared (no <= >= ==) raises."""
          m = Model()
          xs = m.add_variables(2)
          expr = xs[0] + xs[1]
          with pytest.raises(ValueError, match="not a constraint"):
              m.add_constraint(expr)

      def test_valid_constraint_accepted(self):
          m = Model()
          xs = m.add_variables(2)
          m.add_constraint((xs[0] + xs[1]) <= 1)
          assert len(m.con_expr) == 1

      def test_validate_false_skips_none_check(self):
          """validate=False should skip the None check.
          Note: this will likely crash, just verify it doesn't raise ValueError."""
          # We can't pass None with validate=False because it would segfault
          # Instead test with an expression without sense
          m = Model()
          xs = m.add_variables(2)
          expr = xs[0] + xs[1]
          # With validate=False, no ValueError about sense
          # This may or may not work at C level, but Python validation is skipped
          m.add_constraint(expr, validate=False)

  class TestSetObjectiveValidation:

      def test_none_objective_raises(self):
          m = Model()
          with pytest.raises(ValueError, match="cannot be None"):
              m.set_objective(None, MAXIMIZE)

      def test_invalid_sense_raises(self):
          m = Model()
          xs = m.add_variables(2)
          expr = xs[0] + xs[1]
          with pytest.raises(TypeError):
              m.set_objective(expr, sense=999)

      def test_valid_objective_accepted(self):
          m = Model()
          xs = m.add_variables(2)
          expr = xs[0] + xs[1]
          m.set_objective(expr, MAXIMIZE)
          assert m.sense == MAXIMIZE

  class TestCloseValidation:

      def test_empty_model_raises(self):
          """close() on model with no variables raises."""
          m = Model()
          with pytest.raises(ValueError, match="no variables"):
              m.close()

      def test_no_constraints_raises(self):
          """close() on model with variables but no constraints raises."""
          m = Model()
          m.add_variables(3)
          with pytest.raises(ValueError, match="no constraints"):
              m.close()

      def test_valid_model_closes(self):
          m = Model()
          xs = m.add_variables(3)
          m.add_constraint((xs[0] + xs[1] + xs[2]) <= 2)
          m.set_objective(xs[0] + xs[1] + xs[2], MAXIMIZE)
          m.close()
          assert m.constraints_compiled is True

      def test_close_validate_false_skips_checks(self):
          """close(validate=False) skips empty model checks."""
          m = Model()
          xs = m.add_variables(3)
          # No constraints added, but validate=False
          m.set_objective(xs[0] + xs[1] + xs[2], MAXIMIZE)
          m.close(validate=False)
          assert m.constraints_compiled is True

  class TestSolveValidation:

      def test_invalid_results_raises_value_error(self):
          """solve(results='invalid') raises ValueError, not AssertionError."""
          m = Model()
          xs = m.add_variables(3)
          m.add_constraint((xs[0] + xs[1] + xs[2]) <= 2)
          m.set_objective(xs[0] + xs[1] + xs[2], MAXIMIZE)
          m.close()
          with pytest.raises(ValueError, match="results must be"):
              m.solve(stopping_time=1, num_workers=1, results="invalid")

  class TestAddVariablesValidation:

      def test_zero_variables_raises(self):
          m = Model()
          with pytest.raises(ValueError, match=">= 1"):
              m.add_variables(0)

      def test_negative_variables_raises(self):
          m = Model()
          with pytest.raises(ValueError, match=">= 1"):
              m.add_variables(-1)

  class TestDuplicateVariableMerging:

      def test_duplicate_terms_merged_in_constraint(self):
          """3*x0 + 5*x0 should merge to 8*x0 when adding constraint."""
          m = Model()
          xs = m.add_variables(2)
          # Build expression with duplicate variable terms
          expr = 3 * xs[0] + 5 * xs[0] + xs[1]
          m.add_constraint(expr <= 10)
          # If merge works, model should close and solve normally
          m.set_objective(xs[0] + xs[1], MAXIMIZE)
          m.close()
          # No crash = success (the merge happened internally)
