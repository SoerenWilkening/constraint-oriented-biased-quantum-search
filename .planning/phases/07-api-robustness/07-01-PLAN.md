  ```python
  import math

  _INT64_MIN = -(2**63)
  _INT64_MAX = 2**63 - 1

  def _validate_numeric(value, context=""):
      """Validate a numeric value for use as coefficient or constant.

      Rejects NaN, Inf, and values outside int64_t range.
      Called before passing values to C layer.
      """
      if isinstance(value, float):
          if math.isnan(value) or math.isinf(value):
              raise ValueError(
                  f"NaN/Inf not allowed as coefficient{' in ' + context if context else ''}"
              )
          raise TypeError("Float coefficients not allowed; use int")
      if isinstance(value, int) and not isinstance(value, bool):
          if value < _INT64_MIN or value > _INT64_MAX:
              raise OverflowError(
                  f"Coefficient {value} outside int64 range [{_INT64_MIN}, {_INT64_MAX}]"
              )
  ```

              Variable(index=True)

          with pytest.raises(ValueError, match="inconsistent"):
              Variable(index=0, lb=5, ub=0)
