static inline array_t sw_init_arena(size_t B, arena_t *arena) {
    array_t A;
    A.bits = B;
    size_t numbits = (B >> 6) + 1;
    A.n = numbits;
    A.part = (part_length_t*)arena_alloc(arena, A.n * sizeof(part_length_t), 8);
    if (A.part != NULL) {
        memset(A.part, 0, A.n * sizeof(part_length_t));
    }
    return A;
}
