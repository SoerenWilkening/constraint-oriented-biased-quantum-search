```python
"""Problem generators for benchmarking."""

import random


def create_small_problem():
    """Create small benchmark problem.

    Returns dict with:
    - n_vars: number of variables
    - constraints: list of (coeffs, vars, sense, rhs) tuples
    - objective: (coeffs, vars) tuple
    """
    n = 10
    constraints = []
    for i in range(5):
        # Random constraint with 2-4 variables
        n_terms = random.randint(2, 4)
        var_indices = random.sample(range(n), n_terms)
        coeffs = [random.randint(-5, 5) for _ in range(n_terms)]
        sense = random.choice([6, 7, 8])  # GREATER, LOWER, EQUAL
        rhs = random.randint(-10, 10)
        constraints.append((coeffs, var_indices, sense, rhs))

    # Simple objective: sum of first few variables
    obj_vars = list(range(min(5, n)))
    obj_coeffs = [1] * len(obj_vars)

    return {
        'n_vars': n,
        'constraints': constraints,
        'objective': (obj_coeffs, obj_vars)
    }


def create_medium_problem(dense=False):
    """Create medium benchmark problem."""
    n = 50
    n_constraints = 25
    constraints = []

    for i in range(n_constraints):
        # More variables per constraint if dense
        n_terms = random.randint(5, 15) if dense else random.randint(2, 6)
        n_terms = min(n_terms, n)
        var_indices = random.sample(range(n), n_terms)
        coeffs = [random.randint(-10, 10) for _ in range(n_terms)]
        sense = random.choice([6, 7, 8])
        rhs = random.randint(-20, 20)
        constraints.append((coeffs, var_indices, sense, rhs))

    obj_vars = list(range(min(20, n)))
    obj_coeffs = [random.randint(1, 5) for _ in obj_vars]

    return {
        'n_vars': n,
        'constraints': constraints,
        'objective': (obj_coeffs, obj_vars)
    }


def create_large_problem():
    """Create large benchmark problem."""
    n = 200
    n_constraints = 100
    constraints = []

    for i in range(n_constraints):
        n_terms = random.randint(3, 10)
        var_indices = random.sample(range(n), n_terms)
        coeffs = [random.randint(-20, 20) for _ in range(n_terms)]
        sense = random.choice([6, 7, 8])
        rhs = random.randint(-50, 50)
        constraints.append((coeffs, var_indices, sense, rhs))

    obj_vars = list(range(min(50, n)))
    obj_coeffs = [random.randint(1, 10) for _ in obj_vars]

    return {
        'n_vars': n,
        'constraints': constraints,
        'objective': (obj_coeffs, obj_vars)
    }
```

```python
"""Solver performance benchmarks.

Run with: pytest benchmarks/ --benchmark-json=bench_results.json
Compare: pytest benchmarks/ --benchmark-compare --benchmark-compare-fail=mean:20%
"""

import sys
import os

# Add parent directory to path for cbqs import
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

import pytest
from bench_problems import create_small_problem, create_medium_problem, create_large_problem


def setup_model(problem_spec):
    """Create CBQS model from problem specification."""
    from cbqs import Model, Variable

    model = Model()
    n = problem_spec['n_vars']
    variables = [model.add_binary_variable(f"x{i}") for i in range(n)]

    # Add constraints
    for coeffs, var_indices, sense, rhs in problem_spec['constraints']:
        expr = sum(c * variables[v] for c, v in zip(coeffs, var_indices))
        if sense == 6:  # GREATER
            model.add_constraint(expr >= rhs)
        elif sense == 7:  # LOWER
            model.add_constraint(expr <= rhs)
        else:  # EQUAL
            model.add_constraint(expr == rhs)

    # Add objective
    obj_coeffs, obj_vars = problem_spec['objective']
    obj = sum(c * variables[v] for c, v in zip(obj_coeffs, obj_vars))
    model.set_objective(obj, minimize=True)

    return model


class TestSolverBenchmarks:
    """Solver performance benchmarks."""

    def test_bench_small(self, benchmark):
        """Benchmark small problem (10 vars, 5 constraints)."""
        problem = create_small_problem()
        model = setup_model(problem)

        # Warmup
        model.solve(timeout=1.0)

        # Benchmark
        result = benchmark.pedantic(
            lambda: model.solve(timeout=2.0),
            iterations=5,
            rounds=10
        )

        # Basic sanity check
        assert result is not None

    def test_bench_medium(self, benchmark):
        """Benchmark medium problem (50 vars, 25 constraints)."""
        problem = create_medium_problem()
        model = setup_model(problem)

        model.solve(timeout=1.0)

        result = benchmark.pedantic(
            lambda: model.solve(timeout=5.0),
            iterations=3,
            rounds=5
        )

        assert result is not None

    def test_bench_large(self, benchmark):
        """Benchmark large problem (200 vars, 100 constraints)."""
        problem = create_large_problem()
        model = setup_model(problem)

        model.solve(timeout=2.0)

        result = benchmark.pedantic(
            lambda: model.solve(timeout=10.0),
            iterations=2,
            rounds=3
        )

        assert result is not None

    def test_bench_dense(self, benchmark):
        """Benchmark dense constraint problem."""
        problem = create_medium_problem(dense=True)
        model = setup_model(problem)

        model.solve(timeout=1.0)

        result = benchmark.pedantic(
            lambda: model.solve(timeout=5.0),
            iterations=3,
            rounds=5
        )

        assert result is not None

    def test_bench_sparse(self, benchmark):
        """Benchmark sparse constraint problem."""
        problem = create_medium_problem(dense=False)
        model = setup_model(problem)

        model.solve(timeout=1.0)

        result = benchmark.pedantic(
            lambda: model.solve(timeout=5.0),
            iterations=3,
            rounds=5
        )

        assert result is not None


class TestArenaImpact:
    """Tests specifically measuring arena allocation impact."""

    def test_bench_many_iterations(self, benchmark):
        """Benchmark many solver iterations to stress arena.

        The arena optimization should show larger gains on problems
        that require many iterations (more allocation cycles).
        """
        problem = create_medium_problem()
        model = setup_model(problem)

        # Configure for more iterations
        model.set_max_worse_acceptances(50)

        result = benchmark.pedantic(
            lambda: model.solve(timeout=10.0),
            iterations=2,
            rounds=3
        )

        # Extra info for benchmark JSON
        benchmark.extra_info['problem_type'] = 'many_iterations'
        benchmark.extra_info['max_worse_acceptances'] = 50

        assert result is not None
```

```yaml
  benchmark:
    runs-on: ubuntu-latest
    needs: test  # Run after tests pass
    steps:
      - uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.13'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install pytest pytest-benchmark numpy cython
          pip install -e .

      - name: Run benchmarks
        run: |
          pytest benchmarks/ \
            --benchmark-json=benchmark-results.json \
            --benchmark-columns=min,max,mean,stddev,rounds \
            -v

      - name: Upload benchmark results
        uses: actions/upload-artifact@v4
        with:
          name: benchmark-results
          path: benchmark-results.json

      - name: Performance regression check
        if: github.event_name == 'pull_request'
        run: |
          # Download baseline from main branch if available
          # For now, just output results
          echo "Benchmark results:"
          python -c "
          import json
          with open('benchmark-results.json') as f:
              data = json.load(f)
          for bench in data.get('benchmarks', []):
              name = bench['name']
              mean = bench['stats']['mean']
              stddev = bench['stats']['stddev']
              print(f'{name}: {mean:.4f}s (+/- {stddev:.4f}s)')
          "
```
