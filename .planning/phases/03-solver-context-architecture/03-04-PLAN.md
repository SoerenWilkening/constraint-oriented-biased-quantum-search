  <name>Task 3: Update C tests to use ctx instead of global</name>

  <action>
1. Add #include "solver_ctx.h" at the top

2. Update the setup/teardown fixtures:

   Old pattern (uses global):
   static int branching_setup(void **state) {
       memset(&BranchingStats, 0, sizeof(BranchingStats));
       BranchingStats.obj_dependent = NULL;
       BranchingStats.constraint_dependent = NULL;
       return 0;
   }

   New pattern (uses ctx):
   static int branching_setup(void **state) {
       solver_ctx_t *ctx = solver_ctx_create();
       *state = ctx;  // Store ctx for use in tests
       return 0;
   }

   static int branching_teardown(void **state) {
       solver_ctx_t *ctx = (solver_ctx_t *)*state;
       solver_ctx_free(ctx);
       *state = NULL;
       return 0;
   }

3. Update each test to use ctx from **state:

   static void test_set_factors(void **state) {
       solver_ctx_t *ctx = (solver_ctx_t *)*state;
       solver_ctx_set_factors(ctx, 1.0, 2.0, 3.0, 4.0);
       assert_true(ctx->branching_stats.objective_factor == 1.0);
       assert_true(ctx->branching_stats.constraint_factor == 2.0);
       assert_true(ctx->branching_stats.bias_factor == 3.0);
       assert_true(ctx->branching_stats.look_factor == 4.0);
   }
